# v0.8.0 — Auto-Schema & Migration

## Theme

Make OmniQL's core promise actually work: **install, use, switch databases**. Add auto-schema creation so developers don't need external DDL tools, and add cross-driver migration so switching from Postgres to MongoDB (or any source to any dest) is a real, working capability.

## Rationale

Today, `pip install omniql` followed by `engine.insert_one("challenges", {...})` fails on SQL databases because the table doesn't exist. The developer must write raw `CREATE TABLE` SQL — which defeats the purpose of a database-agnostic layer.

This version fixes that fundamental gap:
1. **EnsureTarget** — OmniQL auto-creates tables/collections/indexes before the first write
2. **Schema Inference** — If no schema is registered, infer column types from the document
3. **Migration** — Read all data from source driver, write to dest driver, using OmniQL's own normalized format

After v0.8.0, this workflow works end-to-end:
```python
engine = OmniEngine()
engine.register_postgres_driver("postgres://...")
engine.route("challenges", "postgres")

# Just works — table auto-created from document structure
await engine.insert_one("challenges", {"title": "SQLi 101", "points": 100, "category": "web"})

# Later: switch to MongoDB — zero code changes
engine.register_mongo_driver("mongodb://localhost:27017", "ctfdb")
engine.route("challenges", "mongo")

# Migrate existing data
await engine.migrate("challenges", source="postgres", dest="mongo")
```

## Deliverable 1: New Driver Interface Method — `EnsureTarget`

### Interface Extension (`pkg/core/driver.go`)

```go
// SchemaAwareDriver extends Driver with the ability to create storage targets
// (tables, collections, indexes) automatically.
type SchemaAwareDriver interface {
    Driver

    // EnsureTarget creates the target (table/collection/index) if it doesn't exist.
    // If schema is nil, the driver should accept any document structure (schema-less mode).
    // For SQL drivers, this means CREATE TABLE IF NOT EXISTS with inferred columns.
    // For document/KV drivers, this may be a no-op (auto-create on first write).
    EnsureTarget(ctx context.Context, target string, schema *CollectionSchema) error
}
```

### Engine Auto-Create Logic (`pkg/core/engine.go`)

The engine calls `EnsureTarget` automatically before the first write to a target:

```go
type Engine struct {
    // ... existing fields
    ensuredTargets map[string]bool // tracks which targets have been ensured
}

func (e *Engine) Execute(ctx context.Context, query OQLQuery) (*OmniJSON, error) {
    // ... validation ...
    driver, err := e.selectDriver(query.Target)
    if err != nil {
        return errorResponse(err, "DRIVER_ERROR"), nil
    }

    // Auto-ensure target on first write operation
    if isWriteAction(query.Action) && !e.ensuredTargets[query.Target] {
        if sad, ok := driver.(SchemaAwareDriver); ok {
            schema := e.inferOrGetSchema(query)
            if err := sad.EnsureTarget(ctx, query.Target, schema); err != nil {
                return errorResponse(err, "EXECUTION_ERROR"), nil
            }
            e.ensuredTargets[query.Target] = true
        }
    }

    // ... existing execute logic
}

func isWriteAction(action Action) bool {
    return action == ActionInsert || action == ActionBatchInsert ||
        action == ActionUpdate || action == ActionDelete
}
```

### Schema Inference (`pkg/core/schema.go`)

When no schema is registered, infer from the document:

```go
// InferSchema builds a CollectionSchema from a document's key-value pairs.
// JSON numbers → FieldTypeFloat (JSON has no int/float distinction)
// Strings → FieldTypeString, Booleans → FieldTypeBool
// Nested objects → FieldTypeObject, Arrays → FieldTypeArray
func InferSchema(target string, doc map[string]interface{}) CollectionSchema {
    schema := CollectionSchema{
        Name:   target,
        Fields: make(map[string]FieldSchema),
    }
    for key, val := range doc {
        schema.Fields[key] = FieldSchema{Type: inferFieldType(val)}
    }
    return schema
}

func inferFieldType(val interface{}) FieldType {
    switch val.(type) {
    case string:
        return FieldTypeString
    case float64: // JSON numbers unmarshal as float64
        return FieldTypeFloat
    case bool:
        return FieldTypeBool
    case []interface{}:
        return FieldTypeArray
    case map[string]interface{}:
        return FieldTypeObject
    default:
        return FieldTypeAny
    }
}

// inferOrGetSchema returns the registered schema for a target, or infers
// one from the query's document/documents.
func (e *Engine) inferOrGetSchema(query OQLQuery) *CollectionSchema {
    // Check registered schemas first
    if schema, ok := e.registry.Get(query.Target); ok {
        return &schema
    }

    // Infer from document
    if len(query.Document) > 0 {
        s := InferSchema(query.Target, query.Document)
        return &s
    }
    if len(query.Documents) > 0 {
        // Merge fields from all documents
        merged := make(map[string]interface{})
        for _, doc := range query.Documents {
            for k, v := range doc {
                merged[k] = v
            }
        }
        s := InferSchema(query.Target, merged)
        return &s
    }

    return nil
}
```

## Deliverable 2: SQL Type Mapping for CREATE TABLE

### Type Map (`pkg/drivers/sqlutil/ddl.go`)

Each SQL dialect maps OmniQL field types to native column types differently:

```go
package sqlutil

import "github.com/Uttam-Mahata/omniql/pkg/core"

// TypeMapper maps OmniQL FieldType to a native SQL column type string.
type TypeMapper func(core.FieldType) string

// SQLiteTypes maps FieldType to SQLite column types.
func SQLiteTypes(ft core.FieldType) string {
    switch ft {
    case core.FieldTypeString:  return "TEXT"
    case core.FieldTypeInt:     return "INTEGER"
    case core.FieldTypeFloat:   return "REAL"
    case core.FieldTypeBool:    return "INTEGER"  // SQLite has no BOOLEAN
    case core.FieldTypeArray:   return "TEXT"      // stored as JSON string
    case core.FieldTypeObject:  return "TEXT"      // stored as JSON string
    default:                    return "TEXT"
    }
}

// PostgresTypes maps FieldType to PostgreSQL column types.
func PostgresTypes(ft core.FieldType) string {
    switch ft {
    case core.FieldTypeString:  return "TEXT"
    case core.FieldTypeInt:     return "BIGINT"
    case core.FieldTypeFloat:   return "DOUBLE PRECISION"
    case core.FieldTypeBool:    return "BOOLEAN"
    case core.FieldTypeArray:   return "JSONB"
    case core.FieldTypeObject:  return "JSONB"
    default:                    return "TEXT"
    }
}

// MySQLTypes maps FieldType to MySQL column types.
func MySQLTypes(ft core.FieldType) string {
    switch ft {
    case core.FieldTypeString:  return "TEXT"
    case core.FieldTypeInt:     return "BIGINT"
    case core.FieldTypeFloat:   return "DOUBLE"
    case core.FieldTypeBool:    return "BOOLEAN"
    case core.FieldTypeArray:   return "JSON"
    case core.FieldTypeObject:  return "JSON"
    default:                    return "TEXT"
    }
```

### Auto-ID Column per Dialect

```go
// AutoIDColumn returns the DDL fragment for an auto-incrementing primary key.
func AutoIDColumn(dialect string, quoteIdent func(string) string) string {
    switch dialect {
    case "sqlite":
        return quoteIdent("id") + " INTEGER PRIMARY KEY AUTOINCREMENT"
    case "postgres":
        return quoteIdent("id") + " BIGSERIAL PRIMARY KEY"
    case "mysql":
        return quoteIdent("id") + " BIGINT AUTO_INCREMENT PRIMARY KEY"
    case "sqlserver":
        return quoteIdent("id") + " BIGINT IDENTITY(1,1) PRIMARY KEY"
    default:
        return quoteIdent("id") + " INTEGER PRIMARY KEY"
    }
}
```

### BuildCreateTable

```go
// BuildCreateTable generates a CREATE TABLE IF NOT EXISTS statement.
func BuildCreateTable(
    target string,
    schema *core.CollectionSchema,
    dialect string,
    quoteIdent func(string) string,
    typeMapper TypeMapper,
) string {
    var cols []string

    // Add auto-ID column if schema doesn't define "id"
    if _, hasID := schema.Fields["id"]; !hasID {
        cols = append(cols, AutoIDColumn(dialect, quoteIdent))
    }

    // Sort field names for deterministic output
    keys := sortedKeys(schema.Fields)
    for _, name := range keys {
        field := schema.Fields[name]
        if name == "id" {
            // If schema defines "id", make it the primary key
            cols = append(cols, quoteIdent(name)+" "+typeMapper(field.Type)+" PRIMARY KEY")
            continue
        }
        colDef := quoteIdent(name) + " " + typeMapper(field.Type)
        if field.Required {
            colDef += " NOT NULL"
        }
        cols = append(cols, colDef)
    }

    return fmt.Sprintf("CREATE TABLE IF NOT EXISTS %s (\n  %s\n)",
        quoteIdent(target), strings.Join(cols, ",\n  "))
}
```

## Deliverable 4: Schema Evolution — ALTER TABLE on New Columns

When a developer inserts a document with a field not in the current table, the driver should add the column:

```python
# First insert creates table: challenges(id, title, points)
await engine.insert_one("challenges", {"title": "SQLi", "points": 100})

# Later insert has new field "category" — ALTER TABLE ADD COLUMN
await engine.insert_one("challenges", {"title": "XSS", "points": 200, "category": "web"})
```

### Implementation

```go
// EnsureColumns checks if all columns in the document exist in the table.
// If any are missing, it adds them with ALTER TABLE ADD COLUMN.
func (d *Driver) ensureColumns(ctx context.Context, target string, doc map[string]interface{}) error {
    existing := d.getExistingColumns(ctx, target) // cache this per target
    for key, val := range doc {
        if _, exists := existing[key]; !exists {
            colType := sqlutil.PostgresTypes(core.InferFieldType(val))
            alter := fmt.Sprintf("ALTER TABLE %s ADD COLUMN %s %s",
                quoteIdent(target), quoteIdent(key), colType)
            if _, err := d.db.ExecContext(ctx, alter); err != nil {
                return err
            }
            existing[key] = true
        }
    }
    return nil
}
```

The engine calls this before every INSERT/BATCH_INSERT on SQL drivers. Column info is cached per target to avoid querying `information_schema` repeatedly.

## Deliverable 5: Cross-Driver Migration

### Engine Migration API (`pkg/core/engine.go`)

```go
// MigrateOptions configures a cross-driver migration.
type MigrateOptions struct {
    BatchSize int  // records per BATCH_INSERT (default 100)
    DryRun    bool // count only, don't write
}

// MigrateResult reports migration outcome.
type MigrateResult struct {
    RecordsRead    int64 `json:"records_read"`
    RecordsWritten int64 `json:"records_written"`
    Errors         int   `json:"errors"`
}

// Migrate copies all data from sourceTarget (on its routed driver) to
// destTarget (on its routed driver). The two targets can be on different
// drivers — e.g., migrate "users" from Postgres to MongoDB.
func (e *Engine) Migrate(ctx context.Context, sourceTarget, destTarget string, opts MigrateOptions) (*MigrateResult, error)
```

### How It Works

```
Source Driver (Postgres)                 Dest Driver (MongoDB)
        │                                        │
   FIND target, limit=batchSize, skip=0    →    BATCH_INSERT target
   FIND target, limit=batchSize, skip=100  →    BATCH_INSERT target
   FIND target, limit=batchSize, skip=200  →    BATCH_INSERT target
        │                                        │
   (until FIND returns empty)                (done)
```

The data flows through OmniQL's normalized `[]map[string]interface{}` — no driver-specific code needed. The dest driver's `EnsureTarget` runs automatically on the first BATCH_INSERT.

```go
func (e *Engine) Migrate(ctx context.Context, sourceTarget, destTarget string, opts MigrateOptions) (*MigrateResult, error) {
    if opts.BatchSize <= 0 {
        opts.BatchSize = 100
    }

    srcDriver, err := e.selectDriver(sourceTarget)
    if err != nil {
        return nil, err
    }
    destDriver, err := e.selectDriver(destTarget)
    if err != nil {
        return nil, err
    }

    result := &MigrateResult{}
    skip := 0

    for {
        // Read batch from source
        rows, _, err := srcDriver.Execute(ctx, OQLQuery{
            Target:  sourceTarget,
            Action:  ActionFind,
            Options: QueryOptions{Limit: opts.BatchSize, Skip: skip},
        })
        if err != nil {
            return result, err
        }
        if len(rows) == 0 {
            break // done
        }

        result.RecordsRead += int64(len(rows))

        if !opts.DryRun {
            // Write batch to dest
            written, err := destDriver.BatchInsert(ctx, destTarget, rows)
            if err != nil {
                result.Errors++
                // continue with next batch
            } else {
                result.RecordsWritten += int64(len(written))
            }
        }

        skip += opts.BatchSize
        if len(rows) < opts.BatchSize {
            break // last batch
        }
    }

    return result, nil
}
```

Migrating Data with OmniQL: Postgres to MongoDB

Since OmniQL abstracts the underlying database, the migration process becomes a "Target-to-Target" sync within the same engine.

1. The Migration Config

First, update your config.json to include both the source (Postgres) and the destination (Mongo).

{
  "drivers": {
    "pg_source": { "uri": "postgres://..." },
    "mongo_dest": { "uri": "mongodb+srv://..." }
  },
  "routes": {
    "legacy_challenges": "pg_source",
    "new_challenges": "mongo_dest"
  }
}


2. The Migration Script (Python)

Using the OmniQL SDK, you can pipe data from one driver to the other. The engine handles the type conversions automatically.

import asyncio
from omniql import OmniEngine

async def migrate():
    engine = OmniEngine("config.json")
    
    # 1. Fetch everything from Postgres
    print("Reading from Postgres...")
    pg_result = await engine.execute({
        "target": "legacy_challenges",
        "action": "FIND",
        "filter": {}
    })
    
    rows = pg_result.get("data", [])
    
    # 2. Transform (Optional)
    # This is where you convert flat SQL rows to nested Mongo docs if needed
    for row in rows:
        row["metadata"] = {"migrated_at": "2026-02-26"}
        # OmniQL handles the ID conversion from Serial/UUID to ObjectID
    
    # 3. Batch Insert into MongoDB
    # Optimization: Using BATCH_INSERT reduces network round-trips by 99%
    print(f"Migrating {len(rows)} records to MongoDB...")
    await engine.execute({
        "target": "new_challenges",
        "action": "INSERT_MANY",
        "documents": rows
    })
    
    print("Migration complete!")

if __name__ == "__main__":
    asyncio.run(migrate())


3. Why BATCH_INSERT is Required

Method

Total Time (10k Rows)

Risk Level

Looping Single Inserts

~500 Seconds

High (Partial data on failure)

OmniQL BATCH_INSERT

~2 Seconds

Low (Atomic chunks)

4. The Paradigm Shift (Things to watch)

While OmniQL moves the data, you still need to consider the structural differences:

Relationships: If you had a Foreign Key in Postgres, you should decide if you want to embed that data in the Mongo document or keep it as a reference.

Indexes: You will need to re-create your indexes in MongoDB. OmniQL v0.6.0 plans to include a SYNC_SCHEMA action to automate this, but for now, it should be done in the Mongo console.

## End-to-End Developer Workflow After v0.8.0

### Install and Use (Python FastAPI Example)

```python
# pip install omniql
from omniql import OmniEngine

engine = OmniEngine()
engine.register_postgres_driver("postgres://user:pass@localhost/ctfdb")
engine.route("challenges", "postgres")
engine.route("users", "postgres")
engine.route("submissions", "postgres")

# Just works — tables auto-created
await engine.insert_one("users", {"username": "alice", "score": 0})
await engine.insert_one("challenges", {"title": "SQLi 101", "points": 100, "category": "web"})
await engine.insert_one("submissions", {"user": "alice", "challenge": "SQLi 101", "correct": True})

# Query
web = await engine.find_many("challenges", {"category": "web"})
top = await engine.find_many("users", {"score": {"$gt": 50}}, {"sort": {"score": -1}, "limit": 10})
```

### Switch to MongoDB

```python
# Register new driver
engine.register_mongo_driver("mongodb://localhost:27017", "ctfdb")

# Migrate data
await engine.migrate("challenges", source="postgres", dest="mongo")
await engine.migrate("users", source="postgres", dest="mongo")
await engine.migrate("submissions", source="postgres", dest="mongo")

# Re-route
engine.route("challenges", "mongo")
engine.route("users", "mongo")
engine.route("submissions", "mongo")

# All existing query code works unchanged
web = await engine.find_many("challenges", {"category": "web"})
```

### Hybrid Routing (keep some in Postgres, move some to Redis)

```python
engine.register_redis_driver("redis://localhost:6379/0")
engine.route("sessions", "redis")   # fast cache
engine.route("challenges", "postgres")  # relational data
engine.route("logs", "mongo")       # document store

# Same query API for all three
await engine.insert_one("sessions", {"id": "sess123", "user": "alice", "_ttl": 3600})
await engine.insert_one("challenges", {"title": "XSS", "points": 200})
await engine.insert_one("logs", {"event": "login", "user": "alice"})
```
